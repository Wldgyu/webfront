#include <stdio.h>

int main() {
    int n = 5;

    for (int i = 0; i < n; i++) {
        for (int j = 0; j < 2 * n - 1; j++) {
            if (j == i || j == (2 * n - 2 - i)) {
                printf("*");
            } else {
                printf(" ");
            }
        }
        printf("\n"); // 한 줄 출력이 끝나면 줄바꿈
    }

    return 0;
}
